local function safePath(instance: Instance?): string
	if not instance then return "nil" end
	local ok, result = pcall(function() return instance:GetFullName() end)
	return ok and result or (instance.ClassName .. ":" .. instance.Name)
end

local function containsWord(text: string, words: {string}): boolean
	text = string.lower(text)
	for _, word in ipairs(words) do
		if string.find(text, word, 1, true) then return true end
	end
	return false
end

local function sanitizeString(value: string): string
	value = string.sub(value, 1, CONFIG.MAX_STRING)
	local lower = string.lower(value)
	if string.find(lower, "authorization", 1, true)
		or string.find(lower, "bearer ", 1, true)
		or string.find(lower, "cookie", 1, true)
		or string.find(lower, "token=", 1, true)
		or string.find(lower, "password", 1, true) then
		return "<redacted>"
	end
	return value
end

local function numberText(n: number): string
	if n ~= n then return "nan" end
	if n == math.huge then return "inf" end
	if n == -math.huge then return "-inf" end
	if CONFIG.RAW_PRECISION == "SHORT" then return string.format("%.6g", n) end
	if CONFIG.RAW_PRECISION == "FULL" then return string.format("%.17g", n) end
	return string.format("%.6f", n)
end

local function serialize(value: any, depth: number?, seen: {[table]: boolean}?): string
	depth = depth or 0
	seen = seen or {}
	local kind = typeof(value)
	if kind == "nil" then return "nil" end
	if kind == "string" then return string.format("%q", sanitizeString(value)) end
	if kind == "number" then return numberText(value) end
	if kind == "boolean" then return tostring(value) end
	if kind == "Instance" then return string.format("<%s %s @ %s>", value.ClassName, value.Name, safePath(value)) end
	if kind == "Vector3" then return string.format("Vector3(%s, %s, %s)", numberText(value.X), numberText(value.Y), numberText(value.Z)) end
	if kind == "Vector2" then return string.format("Vector2(%s, %s)", numberText(value.X), numberText(value.Y)) end
	if kind == "Color3" then return string.format("Color3(%s, %s, %s)", numberText(value.R), numberText(value.G), numberText(value.B)) end
	if kind == "CFrame" then
		local parts = {value:GetComponents()}
		for i, part in ipairs(parts) do parts[i] = numberText(part) end
		return "CFrame(" .. table.concat(parts, ", ") .. ")"
	end
	if kind == "PhysicalProperties" then
		return string.format("PhysicalProperties(Density=%s, Friction=%s, Elasticity=%s, FrictionWeight=%s, ElasticityWeight=%s)",
			numberText(value.Density), numberText(value.Friction), numberText(value.Elasticity), numberText(value.FrictionWeight), numberText(value.ElasticityWeight))
	end
	if kind == "EnumItem" then return tostring(value) end
	if kind == "NumberRange" or kind == "NumberSequence" or kind == "ColorSequence" or kind == "UDim" or kind == "UDim2" or kind == "Rect" then
		return tostring(value)
	end
	if kind == "table" then
		if seen[value] then return "<recursive>" end
		if depth >= CONFIG.MAX_DEPTH then return "{...}" end
		seen[value] = true
		local chunks = {}
		local count = 0
		for key, item in pairs(value) do
			count += 1
			if count > CONFIG.MAX_ITEMS then
				table.insert(chunks, "...")
				break
			end
			table.insert(chunks, "[" .. serialize(key, depth + 1, seen) .. "]=" .. serialize(item, depth + 1, seen))
		end
		seen[value] = nil
		return "{" .. table.concat(chunks, ", ") .. "}"
	end
	return "<" .. kind .. ">"
end

local function readProperty(instance: Instance, property: string): (boolean, any)
	return pcall(function() return (instance :: any)[property] end)
end

local function propertiesFor(instance: Instance): {string}
	local output = {}
	if instance:IsA("BasePart") then
		for _, p in ipairs(PROPERTY_MAP.BasePart) do table.insert(output, p) end
	end
	local classProps = PROPERTY_MAP[instance.ClassName]
	if classProps then
		for _, p in ipairs(classProps) do
			if not table.find(output, p) then table.insert(output, p) end
		end
	end
	return output
end

local ConnectionBag = {}
ConnectionBag.__index = ConnectionBag

function ConnectionBag.new()
	return setmetatable({items = {}}, ConnectionBag)
end

function ConnectionBag:add(item: any)
	table.insert(self.items, item)
	return item
end

function ConnectionBag:clear()
	for i = #self.items, 1, -1 do
		local item = self.items[i]
		pcall(function()
			if typeof(item) == "RBXScriptConnection" then item:Disconnect()
			elseif type(item) == "function" then item()
			elseif item.Destroy then item:Destroy() end
		end)
		self.items[i] = nil
	end
end

local function eventSearchText(event: any): string
	return string.lower(event.category .. " " .. event.title .. " " .. event.detail)
end

local function isEventVisible(event: any): boolean
	local group = FILTER_GROUP[event.category] or "OBJECTS"
	if State.filters[group] == false then return false end
	if State.filters.WHEELS == false and containsWord(event.title .. " " .. event.detail, {"wheel", "tire", "tyre", "friction"}) then return false end
	if State.search ~= "" and not string.find(eventSearchText(event), State.search, 1, true) then return false end
	return true
end

	if not UI.logList or not isEventVisible(event) then return end

	row.Text = string.format("[%s] %-9s  %s\n%s", event.stamp, event.category, event.title, event.detail)

local function refreshLog()
	if not UI.logList then return end
	for _, child in ipairs(UI.logList:GetChildren()) do
		if child:IsA("GuiObject") then child:Destroy() end
	end
	for _, event in ipairs(State.events) do
		renderEvent(event)
	end
	task.defer(function()
		if UI.logList then UI.logList.CanvasPosition = Vector2.new(0, math.max(0, UI.logList.AbsoluteCanvasSize.Y)) end
	end)
end

local function correlationSuffix(): string
	local input = State.lastInput
	if not input then return "" end
	local delta = now() - input.time
	if delta < 0 or delta > CONFIG.CORRELATION_WINDOW then return "" end
	return string.format("\n↳ after %s +%dms", input.name, math.floor(delta * 1000 + 0.5))
end

local function addDriftTrace(text: string)
	table.insert(State.driftTrace, text)
	while #State.driftTrace > 80 do table.remove(State.driftTrace, 1) end
	if UI.diffText then
		UI.diffText.Text = (#State.lastDiff > 0 and table.concat(State.lastDiff, "\n") or "No snapshot diff yet.")
			.. "\n\nDRIFT TRACE\n" .. table.concat(State.driftTrace, "\n")
	end
end

local function logEvent(category: string, title: string, detail: string?, source: string?, correlate: boolean?)
	if State.paused and category ~= "INFO" and category ~= "VEHICLE" then return end
	detail = detail or ""
	if correlate then detail ..= correlationSuffix() end
	local t = now() - sessionStart
	local id = string.format("EVT-%04d", State.nextEventId)
	State.nextEventId += 1
	local event = {
		id = id,
		at = t,
		stamp = string.format("+%08.3fs", t),
		category = category,
		title = sanitizeString(title),
		detail = sanitizeString(detail),
		source = source or "INTERNAL",
	}
	table.insert(State.events, event)
	while #State.events > CONFIG.MAX_LOG do
		local removed = table.remove(State.events, 1)
		if UI.logList then
			local oldRow = UI.logList:FindFirstChild(removed.id)
			if oldRow then oldRow:Destroy() end
		end
	end
	local burstEvent = nil
	if correlate and State.lastInput and (now() - State.lastInput.time) <= CONFIG.CORRELATION_WINDOW then
		State.lastInput.count += 1
		if State.lastInput.count == 6 then
			burstEvent = {
				id = string.format("EVT-%04d", State.nextEventId), at=t, stamp=string.format("+%08.3fs", t),
				category="INFO", title="ACTION BURST", detail=State.lastInput.name .. " · 6+ correlated events", source="CORRELATOR"
			}
			table.insert(State.events, burstEvent)
			State.nextEventId += 1
		end
	end
	while #State.events > CONFIG.MAX_LOG do
		local removed = table.remove(State.events, 1)
		if UI.logList then
			local oldRow = UI.logList:FindFirstChild(removed.id)
			if oldRow then oldRow:Destroy() end
		end
	end
	if containsWord(title .. " " .. detail, DRIFT_WORDS) then
		addDriftTrace(string.format("%s  %s  %s — %s", event.stamp, category, title, string.gsub(detail, "\n", " | ")))
	end
	if UI.logList then
		renderEvent(event)
		if burstEvent then renderEvent(burstEvent) end
		task.defer(function()
			if UI.logList then UI.logList.CanvasPosition = Vector2.new(0, math.max(0, UI.logList.AbsoluteCanvasSize.Y)) end
		end)
	end
end

local function categoryFor(instance: Instance): string
	if VALUE_CLASSES[instance.ClassName] then return "VALUE" end
	if PHYSICS_CLASSES[instance.ClassName] then return "PHYSICS" end
	if instance:IsA("Sound") then return "SOUND" end
	if instance:IsA("Animation") or instance:IsA("Animator") then return "ANIMATION" end
	if EFFECT_CLASSES[instance.ClassName] then return "EFFECT" end
	if REMOTE_CLASSES[instance.ClassName] then return "REMOTE" end
	return "PROPERTY"
end

local function isWheel(instance: Instance): boolean
	if not instance:IsA("BasePart") then return false end
	local lower = string.lower(instance.Name)
	if string.find(lower, "physicalwheel", 1, true) then return true end
	if not containsWord(lower, {"wheel", "wheels", "tire", "tyre"}) then return false end
	local score = 1
	if State.connectedParts[instance] then score += 2 end
	if instance.Parent and containsWord(instance.Parent.Name, {"wheel", "suspension", "axle", "chassis", "vehicle"}) then score += 1 end
	for _, child in ipairs(instance:GetChildren()) do
		if child:IsA("Attachment") or child:IsA("Constraint") then score += 1; break end
	end
	return score >= 2
end

local function physicalSnapshot(value: any): any
	if typeof(value) ~= "PhysicalProperties" then return nil end
	return {
		Density=value.Density, Friction=value.Friction, Elasticity=value.Elasticity,
		FrictionWeight=value.FrictionWeight, ElasticityWeight=value.ElasticityWeight,
	}
end

local function initializeWheelMeta(wheel: BasePart): any
	local meta = State.wheelMeta[wheel]
	if meta then return meta end
	meta = {
		currentCustom = physicalSnapshot(wheel.CustomPhysicalProperties),
		currentCurrent = physicalSnapshot(wheel.CurrentPhysicalProperties),
		previousCustom = nil,
		baselineCustom = nil,
		baselineCurrent = nil,
		history = {},
		attachments = {},
		constraints = {},
		changes = 0,
	}
	State.wheelMeta[wheel] = meta
	return meta
end

local function refreshWheelRelations()
	for wheel in pairs(State.wheels) do
		if wheel.Parent and wheel:IsA("BasePart") then
			local meta = initializeWheelMeta(wheel)
			table.clear(meta.attachments)
			table.clear(meta.constraints)
			for instance in pairs(State.cache) do
				if instance:IsA("Attachment") and instance:IsDescendantOf(wheel) then
					table.insert(meta.attachments, instance)
				elseif instance:IsA("Constraint") or instance:IsA("WeldConstraint") or instance:IsA("Motor6D") then
					local related = instance:IsDescendantOf(wheel)
					for _, property in ipairs({"Attachment0", "Attachment1", "Part0", "Part1"}) do
						local ok, linked = readProperty(instance, property)
						if ok and typeof(linked) == "Instance" and (linked == wheel or linked:IsDescendantOf(wheel)) then related = true end
					end
					if related then table.insert(meta.constraints, instance) end
				end
			end
		end
	end
end

local function onWheelPhysicalChanged(wheel: BasePart, oldValue: any, newValue: any)
	local meta = initializeWheelMeta(wheel)
	meta.previousCustom = physicalSnapshot(oldValue)
	meta.currentCustom = physicalSnapshot(newValue)
	meta.currentCurrent = physicalSnapshot(wheel.CurrentPhysicalProperties)
	meta.changes += 1
	local oldFriction = meta.previousCustom and meta.previousCustom.Friction or nil
	local newFriction = meta.currentCustom and meta.currentCustom.Friction or nil
	table.insert(meta.history, {
		at=now()-sessionStart,
		old=oldFriction,
		new=newFriction,
	})
	while #meta.history > 20 do table.remove(meta.history, 1) end
	if not State.freezeView and UI.flashWheel then UI.flashWheel(wheel) end
	if UI.updateWheel then UI.updateWheel(wheel, false) end
end

local function formatPhysicalChange(old: any, new: any): string
	if typeof(old) ~= "PhysicalProperties" or typeof(new) ~= "PhysicalProperties" then
		return serialize(old) .. " → " .. serialize(new)
	end
	local rows = {}
	local keys = {"Density", "Friction", "Elasticity", "FrictionWeight", "ElasticityWeight"}
	local oldSnapshot, newSnapshot = physicalSnapshot(old), physicalSnapshot(new)
	for _, key in ipairs(keys) do
		local a, b = oldSnapshot[key], newSnapshot[key]
		if a ~= b then table.insert(rows, string.upper(key) .. "  " .. numberText(a) .. " → " .. numberText(b)) end
	end
	return #rows > 0 and table.concat(rows, "\n") or (serialize(old) .. " → " .. serialize(new))
end

local function disconnectInstance(instance: Instance)
	local bag = instanceConnections[instance]
	if bag then bag:clear() end
	instanceConnections[instance] = nil
	State.cache[instance] = nil
	State.wheels[instance] = nil
	State.wheelMeta[instance] = nil
	State.remotes[instance] = nil
end

local function watchInstance(instance: Instance, baseline: boolean)
	if State.cache[instance] then return end
	local record = {
		class = instance.ClassName,
		name = instance.Name,
		path = safePath(instance),
		attributes = instance:GetAttributes(),
		properties = {},
		baseline = baseline,
		changes = 0,
	}
	State.cache[instance] = record
	if isWheel(instance) then
		State.wheels[instance] = true
		initializeWheelMeta(instance :: BasePart)
	end
	if REMOTE_CLASSES[instance.ClassName] then
		State.remotes[instance] = {path=record.path, class=instance.ClassName, baseline=baseline, lastCall="None"}
	end
	local bag = ConnectionBag.new()
	instanceConnections[instance] = bag

	if VALUE_CLASSES[instance.ClassName] then
		local ok, value = readProperty(instance, "Value")
		if ok then record.properties.Value = value end
		bag:add(instance:GetPropertyChangedSignal("Value"):Connect(function()
			local old = record.properties.Value
			local okNow, new = readProperty(instance, "Value")
			if okNow and old ~= new then
				record.properties.Value = new
				record.changes += 1
				local label = baseline and "VALUE CHANGED" or "NEW VALUE"
				logEvent("VALUE", label .. " · " .. instance.Name, serialize(old) .. " → " .. serialize(new) .. "\n" .. safePath(instance), "PROPERTY SIGNAL", true)
			end
		end))
	end

	for _, property in ipairs(propertiesFor(instance)) do
		local ok, value = readProperty(instance, property)
		if ok then
			record.properties[property] = value
			local signalOk, signal = pcall(function() return instance:GetPropertyChangedSignal(property) end)
			if signalOk then
				bag:add(signal:Connect(function()
					local old = record.properties[property]
					local nowOk, new = readProperty(instance, property)
					if not nowOk or old == new then return end
					record.properties[property] = new
					record.changes += 1
					local category = categoryFor(instance)
					local detail = serialize(old) .. " → " .. serialize(new) .. "\n" .. safePath(instance)
					if property == "CustomPhysicalProperties" then
						category = "PHYSICS"
						detail = formatPhysicalChange(old, new) .. "\n" .. safePath(instance)
						if State.wheels[instance] and instance:IsA("BasePart") then onWheelPhysicalChanged(instance, old, new) end
					end
					local title = instance.Name .. "." .. property
					if isWheel(instance) and (property == "CustomPhysicalProperties" or containsWord(title, DRIFT_WORDS)) then
						title = "DRIFT CANDIDATE · " .. title
					end
					logEvent(category, title, detail, "PROPERTY SIGNAL", true)
					if (property == "Attachment0" or property == "Attachment1" or property == "Part0" or property == "Part1") and UI.rebuildWheels then
						refreshWheelRelations(); UI.rebuildWheels()
					end
				end))
			end
		end
	end

	bag:add(instance.AttributeChanged:Connect(function(attributeName)
		local old = record.attributes[attributeName]
		local new = instance:GetAttribute(attributeName)
		if old == new then return end
		record.attributes[attributeName] = new
		record.changes += 1
		local prefix = old == nil and "NEW ATTRIBUTE · " or ""
		local title = prefix .. instance.Name .. "." .. attributeName
		if containsWord(title, DRIFT_WORDS) then title = "DRIFT CANDIDATE · " .. title end
		logEvent("ATTRIBUTE", title, serialize(old) .. " → " .. serialize(new) .. "\n" .. safePath(instance), "ATTRIBUTE SIGNAL", true)
		if UI.rebuildDataLists and (old == nil or new == nil) then UI.rebuildDataLists()
		elseif UI.updateDataLists then UI.updateDataLists() end
	end))

	if instance:IsA("Animator") then
		bag:add(instance.AnimationPlayed:Connect(function(track)
			local animationId = "unknown"
			pcall(function() animationId = track.Animation.AnimationId end)
			logEvent("ANIMATION", "TRACK PLAYED", instance.Name .. "\nAnimationId: " .. animationId, "ANIMATION SIGNAL", true)
		end))
	end

	if instance:IsA("RemoteEvent") or instance:IsA("UnreliableRemoteEvent") then
		bag:add(instance.OnClientEvent:Connect(function(...)
			local args = table.pack(...)
			local printable = {}
			for i = 1, math.min(args.n, CONFIG.MAX_ITEMS) do
				table.insert(printable, string.format("[%d] %s", i, serialize(args[i])))
			end
			State.lastRemote = instance.Name .. " (IN)"
			logEvent("REMOTE", "REMOTE IN · " .. instance.Name, table.concat(printable, "\n") .. "\n" .. safePath(instance), "ON CLIENT EVENT", true)
		end))
	end

	if not baseline and State.currentVehicle then
		local detail = instance.ClassName .. "\nPath: " .. record.path
		if IMPORTANT_CLASSES[instance.ClassName] then detail = "NEW / IMPORTANT\n" .. detail end
		logEvent("OBJECT+", "NEW DISCOVERY · " .. instance.Name, detail, "DESCENDANT ADDED", true)
		if UI.rebuildTree then UI.rebuildTree() end
		if UI.rebuildDataLists and (VALUE_CLASSES[instance.ClassName] or next(instance:GetAttributes()) ~= nil) then UI.rebuildDataLists() end
		if UI.rebuildWheels and (State.wheels[instance] or instance:IsA("Attachment") or instance:IsA("Constraint") or instance:IsA("WeldConstraint") or instance:IsA("Motor6D")) then
			refreshWheelRelations()
			UI.rebuildWheels()
		end
	end
end

local function modelScore(model: Model, seat: BasePart, connected: {[BasePart]: boolean}): number
	if model == workspace or model == workspace.CurrentCamera then return -math.huge end
	local lower = string.lower(model.Name)
	if lower == "workspace" or containsWord(lower, {"map", "building", "garage", "terrain"}) then return -250 end
	local score = 0
	if seat:IsDescendantOf(model) then score += 20 end
	local descendants = model:GetDescendants()
	if #descendants > 1500 then score -= 100 end
	local parts, seats, wheels, connectedInside = 0, 0, 0, 0
	for _, item in ipairs(descendants) do
		if item:IsA("BasePart") then
			parts += 1
			if connected[item] then connectedInside += 1 end
			if isWheel(item) then wheels += 1 end
			if item:IsA("VehicleSeat") then seats += 1 end
		end
	end
	score += math.min(parts, 80) * 0.08 + wheels * 7 + seats * 5 + connectedInside * 1.5
	if containsWord(model.Name, {"vehicle", "car", "truck", "bike", "motor", "suv", "van"}) then score += 12 end
	if model.PrimaryPart then score += 2 end
	return score
end

local function resolveVehicle(seat: Seat): Instance?
	local connected: {[BasePart]: boolean} = {}
	local ok, parts = pcall(function() return seat:GetConnectedParts(true) end)
	if ok then for _, part in ipairs(parts) do connected[part] = true end end
	connected[seat] = true
	local assemblyRoot = seat.AssemblyRootPart
	if assemblyRoot then connected[assemblyRoot] = true end
	local candidates = {}
	local ancestor = seat.Parent
	while ancestor and ancestor ~= workspace do
		if ancestor:IsA("Model") then table.insert(candidates, ancestor) end
		ancestor = ancestor.Parent
	end
	for part in pairs(connected) do
		local model = part:FindFirstAncestorOfClass("Model")
		if model and not table.find(candidates, model) then table.insert(candidates, model) end
	end
	local best: Model? = nil
	local bestScore = -math.huge
	for _, model in ipairs(candidates) do
		local score = modelScore(model, seat, connected)
		if score > bestScore then best, bestScore = model, score end
	end
	if best and bestScore > 0 then return best end
	return seat.Parent
end

local function inventorySummary(root: Instance): string
	local counts = {Instances=0, Models=0, Parts=0, Values=0, Attributes=0, Remotes=0, Constraints=0, Effects=0, Sounds=0, Animations=0, Attachments=0, Motors=0, Welds=0}
	local all = {root}
	for _, item in ipairs(root:GetDescendants()) do table.insert(all, item) end
	for _, item in ipairs(all) do
		counts.Instances += 1
		for _ in pairs(item:GetAttributes()) do counts.Attributes += 1 end
		if item:IsA("Model") then counts.Models += 1 end
		if item:IsA("BasePart") then counts.Parts += 1 end
		if item:IsA("ValueBase") then counts.Values += 1 end
		if REMOTE_CLASSES[item.ClassName] then counts.Remotes += 1 end
		if item:IsA("Constraint") then counts.Constraints += 1 end
		if EFFECT_CLASSES[item.ClassName] then counts.Effects += 1 end
		if item:IsA("Sound") then counts.Sounds += 1 end
		if item:IsA("Animation") or item:IsA("Animator") then counts.Animations += 1 end
		if item:IsA("Attachment") then counts.Attachments += 1 end
		if item:IsA("Motor6D") or item:IsA("Motor") then counts.Motors += 1 end
		if item:IsA("Weld") or item:IsA("WeldConstraint") then counts.Welds += 1 end
	end
	return string.format("Instances: %d\nModels: %d  Parts: %d\nValues: %d  Attributes: %d\nRemotes: %d  Constraints: %d\nEffects: %d  Sounds: %d\nAnimations: %d  Attachments: %d\nMotors: %d  Welds: %d",
		counts.Instances, counts.Models, counts.Parts, counts.Values, counts.Attributes, counts.Remotes, counts.Constraints,
		counts.Effects, counts.Sounds, counts.Animations, counts.Attachments, counts.Motors, counts.Welds)
end

local function startVehicle(seat: Seat)
	local vehicle = resolveVehicle(seat)
	if not vehicle then
		logEvent("WARNING", "VEHICLE RESOLVER", "No safe vehicle root found.", "RESOLVER")
		return
	end
	if State.currentVehicle == vehicle and State.currentSeat == seat then return end
	if State.currentVehicle then logEvent("VEHICLE", "STOP OLD VEHICLE", State.vehicleId, "SEAT CHANGE") end
	clearVehicleConnections()
	scanRelevantRemotes()
	State.currentSeat = seat
	State.currentVehicle = vehicle
	local connectedOk, connected = pcall(function() return seat:GetConnectedParts(true) end)
	if connectedOk then for _, part in ipairs(connected) do State.connectedParts[part] = true end end
	State.connectedParts[seat] = true
	if seat.AssemblyRootPart then State.connectedParts[seat.AssemblyRootPart] = true end
	State.vehicleId = vehicle.Name .. "-" .. HttpService:GenerateGUID(false):sub(1, 8)
	State.lastSnapshot = nil
	State.lastDiff = {}
	State.driftTrace = {}
	if UI.vehicleLabel then UI.vehicleLabel.Text = "VEHICLE  " .. vehicle.Name end
	if UI.statusLabel then UI.statusLabel.Text = "● MONITORING"; UI.statusLabel.TextColor3 = Color3.fromRGB(96, 220, 146) end
	logEvent("VEHICLE", "VEHICLE FOUND", vehicle.Name .. "\nSeat: " .. safePath(seat) .. "\nRoot: " .. safePath(vehicle) .. "\nID: " .. State.vehicleId, "VEHICLE RESOLVER")

	local all = {vehicle}
	for _, item in ipairs(vehicle:GetDescendants()) do table.insert(all, item) end
	for _, item in ipairs(all) do watchInstance(item, true) end
	refreshWheelRelations()
	logEvent("INFO", "VEHICLE SNAPSHOT", inventorySummary(vehicle), "INITIAL SCAN")

	vehicleConnections:add(vehicle.DescendantAdded:Connect(function(item)
		watchInstance(item, false)
	end))
	vehicleConnections:add(vehicle.DescendantRemoving:Connect(function(item)
		local record = State.cache[item]
		if record then
			logEvent("OBJECT-", item.Name, item.ClassName .. "\nPath: " .. record.path, "DESCENDANT REMOVING", true)
			disconnectInstance(item)
			if UI.rebuildTree then UI.rebuildTree() end
			if UI.rebuildDataLists then UI.rebuildDataLists() end
			if UI.rebuildWheels then refreshWheelRelations(); UI.rebuildWheels() end
		end
	end))
	if UI.rebuildTree then UI.rebuildTree() end
	if UI.rebuildDataLists then UI.rebuildDataLists() end
	if UI.rebuildWheels then UI.rebuildWheels() end
	if UI.refreshRemotes then UI.refreshRemotes() end
end

local function stopVehicle()
	if not State.currentVehicle then return end
	local old = State.currentVehicle
	clearVehicleConnections()
	State.currentSeat = nil
	State.currentVehicle = nil
	State.vehicleId = "NONE"
	if UI.vehicleLabel then UI.vehicleLabel.Text = "VEHICLE  NONE" end
	if UI.statusLabel then UI.statusLabel.Text = "● WAITING"; UI.statusLabel.TextColor3 = Color3.fromRGB(255, 194, 92) end
	logEvent("VEHICLE", "VEHICLE MONITOR PAUSED", old.Name, "SEAT CHANGE")
	if UI.rebuildTree then UI.rebuildTree() end
	if UI.rebuildDataLists then UI.rebuildDataLists() end
	if UI.rebuildWheels then UI.rebuildWheels() end
	if UI.refreshRemotes then UI.refreshRemotes() end
end

local function captureSnapshot(): any
	local snapshot = {at=now(), vehicle=State.currentVehicle, items={}}
	for instance, record in pairs(State.cache) do
		if not State.currentVehicle or (instance ~= State.currentVehicle and not instance:IsDescendantOf(State.currentVehicle)) then continue end
		local item = {name=instance.Name, class=instance.ClassName, path=safePath(instance), attributes={}, properties={}}
		for key, value in pairs(instance:GetAttributes()) do item.attributes[key] = serialize(value) end
		if VALUE_CLASSES[instance.ClassName] then
			local ok, value = readProperty(instance, "Value")
			if ok then item.properties.Value = serialize(value) end
		end
		for _, property in ipairs(propertiesFor(instance)) do
			local ok, value = readProperty(instance, property)
			if ok then item.properties[property] = serialize(value) end
		end
		snapshot.items[instance] = item
	end
	return snapshot
end

local function compareSnapshots(before: any, after: any): {string}
	local lines = {}
	local changed = 0
	for instance, old in pairs(before.items) do
		local new = after.items[instance]
		if not new then
			changed += 1
			table.insert(lines, "REMOVED  " .. old.path .. " [" .. old.class .. "]")
		else
			for key, oldValue in pairs(old.attributes) do
				local newValue = new.attributes[key]
				if oldValue ~= newValue then changed += 1; table.insert(lines, new.path .. ".@" .. key .. "\n  " .. oldValue .. " → " .. tostring(newValue)) end
			end
			for key, newValue in pairs(new.attributes) do
				if old.attributes[key] == nil then changed += 1; table.insert(lines, "NEW ATTRIBUTE  " .. new.path .. ".@" .. key .. " = " .. newValue) end
			end
			for key, oldValue in pairs(old.properties) do
				local newValue = new.properties[key]
				if oldValue ~= newValue then changed += 1; table.insert(lines, new.path .. "." .. key .. "\n  " .. oldValue .. " → " .. tostring(newValue)) end
			end
		end
	end
	for instance, new in pairs(after.items) do
		if not before.items[instance] then changed += 1; table.insert(lines, "ADDED  " .. new.path .. " [" .. new.class .. "]") end
	end
	table.insert(lines, 1, "SNAPSHOT DIFF\nCHANGED  " .. changed .. "\n")
	return lines
end

local function snapshotAction()
	if not State.currentVehicle then logEvent("WARNING", "SNAPSHOT", "Enter a vehicle first.", "UI"); return end
	local current = captureSnapshot()
	if not State.lastSnapshot or State.lastSnapshot.vehicle ~= State.currentVehicle then
		State.lastSnapshot = current
		State.lastDiff = {"SNAPSHOT A SAVED", string.format("+%.3fs", current.at - sessionStart), tostring(#(State.currentVehicle:GetDescendants()) + 1) .. " instances"}
		logEvent("INFO", "SNAPSHOT A SAVED", "Press SNAPSHOT again to compare.", "SNAPSHOT MANAGER")
	else
		State.lastDiff = compareSnapshots(State.lastSnapshot, current)
		State.lastSnapshot = current
		logEvent("INFO", "SNAPSHOT COMPARED", State.lastDiff[1] .. "\nBaseline advanced to current state.", "SNAPSHOT MANAGER")
	end
	if UI.diffText then UI.diffText.Text = table.concat(State.lastDiff, "\n") .. "\n\nDRIFT TRACE\n" .. table.concat(State.driftTrace, "\n") end
end

local function appendReadableProperties(lines: {string}, instance: Instance, properties: {string}, indent: string?)
	indent = indent or ""
	for _, property in ipairs(properties) do
		local ok, value = readProperty(instance, property)
		if ok then table.insert(lines, indent .. property .. "  " .. serialize(value)) end
	end

local function wheelRole(wheel: BasePart): string
	local seat = State.currentSeat
	if not seat or not seat.Parent then return wheel.Name end
	local relative = seat.CFrame:PointToObjectSpace(wheel.Position)
	local longitudinal = relative.Z <= 0 and "F" or "R"
	local lateral = relative.X <= 0 and "L" or "R"
	return longitudinal .. lateral
end

local function physicalLines(title: string, physical: any): {string}
	local lines = {title}
	if not physical then
		table.insert(lines, "  NONE / MATERIAL DEFAULT")
		return lines
	end
	for _, key in ipairs(PHYSICAL_KEYS) do
		table.insert(lines, string.format("  %-17s %s", key, numberText(physical[key])))
	end
	return lines
end

local function wheelCardText(wheel: BasePart): string
	local meta = initializeWheelMeta(wheel)
	meta.currentCustom = physicalSnapshot(wheel.CustomPhysicalProperties)
	meta.currentCurrent = physicalSnapshot(wheel.CurrentPhysicalProperties)
	local lines = {
		string.format("[%s] %s", wheelRole(wheel), wheel.Name),
		"ClassName  " .. wheel.ClassName,
		"Path       " .. safePath(wheel),
		"",
	}
	for _, line in ipairs(physicalLines("CUSTOM PHYSICAL PROPERTIES", meta.currentCustom)) do table.insert(lines, line) end
	table.insert(lines, "")
	for _, line in ipairs(physicalLines("CURRENT PHYSICAL PROPERTIES", meta.currentCurrent)) do table.insert(lines, line) end
	local currentFriction = meta.currentCustom and meta.currentCustom.Friction or meta.currentCurrent.Friction
	local previousFriction = meta.previousCustom and meta.previousCustom.Friction or nil
	local baselinePhysical = meta.baselineCustom or meta.baselineCurrent
	local baselineFriction = baselinePhysical and baselinePhysical.Friction or nil
	table.insert(lines, "")
	table.insert(lines, "COMPARISON")
	table.insert(lines, "  CURRENT   " .. numberText(currentFriction))
	table.insert(lines, "  PREVIOUS  " .. (previousFriction and numberText(previousFriction) or "—"))
	table.insert(lines, "  DELTA     " .. (previousFriction and numberText(currentFriction - previousFriction) or "—"))
	table.insert(lines, "  BASELINE  " .. (baselineFriction and numberText(baselineFriction) or "—"))
	table.insert(lines, "  BASE Δ    " .. (baselineFriction and numberText(currentFriction - baselineFriction) or "—"))
	table.insert(lines, "")
	appendReadableProperties(lines, wheel, WHEEL_LIVE_PROPERTIES, "  ")
	table.insert(lines, "")
	table.insert(lines, "ATTACHMENTS  [" .. #meta.attachments .. "]")
	for _, attachment in ipairs(meta.attachments) do
		table.insert(lines, "  " .. attachment.Name)
		appendReadableProperties(lines, attachment, {"Position", "Orientation", "Axis", "SecondaryAxis"}, "    ")
	end
	table.insert(lines, "")
	table.insert(lines, "CONSTRAINTS  [" .. #meta.constraints .. "]")
	for _, constraint in ipairs(meta.constraints) do
		table.insert(lines, "  " .. constraint.ClassName .. "  " .. constraint.Name)
		appendReadableProperties(lines, constraint, CONSTRAINT_DETAILS[constraint.ClassName] or propertiesFor(constraint), "    ")
	end
	table.insert(lines, "")
	table.insert(lines, "PHYSICAL PROPERTY HISTORY  [last 20]")
	for _, entry in ipairs(meta.history) do
		table.insert(lines, string.format("  +%08.3fs  %s → %s", entry.at, entry.old and numberText(entry.old) or "nil", entry.new and numberText(entry.new) or "nil"))
	end
	return table.concat(lines, "\n")
end

		"Path: " .. safePath(instance),
		"Parent: " .. (instance.Parent and safePath(instance.Parent) or "nil"),

		for _, property in ipairs(WHEEL_LIVE_PROPERTIES) do if not table.find(properties, property) then table.insert(properties, property) end end

		local deepSets = {}

	for _, child in ipairs(children) do table.insert(lines, "  " .. child.ClassName .. "  " .. child.Name) end

local function comparisonReport(): string
	local wheels = {}
	for wheel in pairs(State.wheels) do if wheel.Parent and wheel:IsA("BasePart") then table.insert(wheels, wheel) end end
	table.sort(wheels, function(a, b) return wheelRole(a) .. a.Name < wheelRole(b) .. b.Name end)
	local lines = {"COMPARE ALL WHEELS", "RAW PRECISION  " .. CONFIG.RAW_PRECISION, ""}
	local header = string.format("%-18s", "PROPERTY")
	for _, wheel in ipairs(wheels) do header ..= string.format("%-15s", wheelRole(wheel)) end
	table.insert(lines, header)
	local difference = false
	for _, key in ipairs(PHYSICAL_KEYS) do
		local row = string.format("%-18s", key)
		local first = nil
		for _, wheel in ipairs(wheels) do
			local current = physicalSnapshot(wheel.CustomPhysicalProperties) or physicalSnapshot(wheel.CurrentPhysicalProperties)
			local value = current and current[key] or nil
			if first == nil then first = value elseif first ~= value then difference = true end
			row ..= string.format("%-15s", value and numberText(value) or "nil")
		end
		table.insert(lines, row)
	end
	if difference then
		table.insert(lines, 2, "DIFFERENCE DETECTED")
		local front,rear={},{}
		for _, wheel in ipairs(wheels) do
			local current=physicalSnapshot(wheel.CustomPhysicalProperties) or physicalSnapshot(wheel.CurrentPhysicalProperties)
			local target=string.sub(wheelRole(wheel),1,1)=="F" and front or rear
			if current then table.insert(target,wheelRole(wheel).."="..numberText(current.Friction)) end
		end
		table.insert(lines,3,"Front: "..(#front>0 and table.concat(front,"  ") or "—").."\nRear:  "..(#rear>0 and table.concat(rear,"  ") or "—"))
	end
	table.insert(lines, "\nDRIFT TRACE")
	for i = math.max(1, #State.driftTrace - 19), #State.driftTrace do table.insert(lines, State.driftTrace[i]) end
	return table.concat(lines, "\n")
end

local function finishSPAToolsIntro(screen, startedAt)
	if not screen or not screen.Parent then return end

	-- Show the complete greeting briefly, without imposing a long loading delay.
	local elapsed = os.clock() - startedAt
	if elapsed < 1.22 then task.wait(1.22 - elapsed) end
	if not screen.Parent then return end

	local background = screen:FindFirstChild("Background")
	if not background then screen:Destroy(); return end

	local fadeInfo = TweenInfo.new(0.42, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
	local title = background:FindFirstChild("Title")
	local accentLine = background:FindFirstChild("AccentLine")
	local glowLine = background:FindFirstChild("GlowLine")
	local welcomeRow = background:FindFirstChild("WelcomeRow")

	TweenService:Create(background, fadeInfo, {BackgroundTransparency = 1}):Play()
	if title then
		TweenService:Create(title, fadeInfo, {
			Position = UDim2.new(0.5, 0, 0.43, -9),
			TextTransparency = 1,
			TextStrokeTransparency = 1,
		}):Play()
	end
	if accentLine then
		TweenService:Create(accentLine, fadeInfo, {
			Size = UDim2.fromOffset(0, 2),
			BackgroundTransparency = 1,
		}):Play()
	end
	if glowLine then
		TweenService:Create(glowLine, fadeInfo, {
			Size = UDim2.fromOffset(0, 8),
			BackgroundTransparency = 1,
		}):Play()
	end
	if welcomeRow then
		TweenService:Create(welcomeRow, fadeInfo, {
			Position = UDim2.new(0.5, 0, 0.43, 112),
		}):Play()
		local avatarRing = welcomeRow:FindFirstChild("AvatarRing")
		local greeting = welcomeRow:FindFirstChild("Greeting")
		local usernameLabel = welcomeRow:FindFirstChild("Username")
		if avatarRing then
			TweenService:Create(avatarRing, fadeInfo, {BackgroundTransparency = 1}):Play()
			local avatarImage = avatarRing:FindFirstChild("AvatarImage")
			local avatarFallback = avatarRing:FindFirstChild("AvatarFallback")
			if avatarImage then
				TweenService:Create(avatarImage, fadeInfo, {
					ImageTransparency = 1,
					BackgroundTransparency = 1,
				}):Play()
			end
			if avatarFallback then
				TweenService:Create(avatarFallback, fadeInfo, {TextTransparency = 1}):Play()
			end
		end
		if greeting then TweenService:Create(greeting, fadeInfo, {TextTransparency = 1}):Play() end
		if usernameLabel then TweenService:Create(usernameLabel, fadeInfo, {TextTransparency = 1}):Play() end
	end

	task.wait(0.45)
	if screen.Parent then screen:Destroy() end
end

local function inputName(input: InputObject): string
	if input.KeyCode ~= Enum.KeyCode.Unknown then return input.KeyCode.Name end
	return input.UserInputType.Name
end

local function shouldLogInput(input: InputObject): boolean
	if UserInputService:GetFocusedTextBox() then return false end
	if input.UserInputType == Enum.UserInputType.Keyboard then return true end
	return input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.MouseButton2
		or input.UserInputType == Enum.UserInputType.Gamepad1
		or input.UserInputType == Enum.UserInputType.Touch
end

local function bindCharacter(character: Model)
	local humanoid = character:WaitForChild("Humanoid", 10)
	if not humanoid or not humanoid:IsA("Humanoid") then return end
	local function updateSeat()
		local seat = humanoid.SeatPart
		if seat and (seat:IsA("VehicleSeat") or seat:IsA("Seat")) then startVehicle(seat) else stopVehicle() end
	end
	permanentConnections:add(humanoid:GetPropertyChangedSignal("SeatPart"):Connect(updateSeat))
	updateSeat()
end

logEvent("INFO", "SPA TOOLS CAPABILITIES", "Vehicle Tree      ✓\nProperties        ✓\nAttributes        ✓\nPhysics           ✓\nInputs            ✓\nRemote Incoming   ✓\nRemote Outgoing   ✗ (standard client API unavailable)\nClipboard         ✗ (manual selection only)", "SPA TOOLS")

task.spawn(function()
	while State.running and UI.screen and UI.screen.Parent do
		task.wait(CONFIG.TELEMETRY_INTERVAL)
		local vehicle = State.currentVehicle
		local seat = State.currentSeat
		if vehicle and seat and seat.Parent then
			local velocity = seat.AssemblyLinearVelocity
			local angular = seat.AssemblyAngularVelocity
			local wheelRows = {}
			local activeForces, activeEffects = 0, 0
			for instance in pairs(State.cache) do
				if instance.Parent then
					if PHYSICS_CLASSES[instance.ClassName] then activeForces += 1 end
					if EFFECT_CLASSES[instance.ClassName] then
						local ok, enabled = readProperty(instance, "Enabled")
						if ok and enabled then activeEffects += 1 end
					end
				end
			end
			for wheel in pairs(State.wheels) do
				if wheel.Parent and wheel:IsA("BasePart") then
					local custom = wheel.CustomPhysicalProperties
					local current = wheel.CurrentPhysicalProperties
					local friction = custom and custom.Friction or current.Friction
					table.insert(wheelRows, wheel.Name .. "=" .. numberText(friction))
				end
			end
			local active = {}
			for instance, record in pairs(State.cache) do
				if record.changes and record.changes > 0 and (instance == vehicle or instance:IsDescendantOf(vehicle)) then table.insert(active,{instance=instance,count=record.changes}) end
			end
			table.sort(active,function(a,b) return a.count>b.count end)
			local activityRows = {}
			for i=1,math.min(5,#active) do table.insert(activityRows,string.format("  %d. %s  %d",i,active[i].instance.Name,active[i].count)) end
			if #activityRows==0 then table.insert(activityRows,"  No changes yet") end
			if UI.liveText then
				UI.liveText.Text = string.format("VEHICLE       %s\nSEAT          %s\nSPEED RAW     %s studs/s\nLIN VEL       %s\nANG VEL       %s\nWHEEL FRICTION\n  %s\nACTIVE FORCES %d\nACTIVE EFFECTS %d\nLAST INPUT    %s\nLAST REMOTE   %s\nRAW PRECISION %s\n\n%s\n\nACTIVITY · TOP CHANGING OBJECTS\n%s",
					vehicle.Name, seat.Name, numberText(velocity.Magnitude), serialize(velocity), serialize(angular),
					table.concat(wheelRows, "  "), activeForces, activeEffects,
					State.lastInput and State.lastInput.name or "None", State.lastRemote, CONFIG.RAW_PRECISION,
					vehicleSeatReport(seat), table.concat(activityRows,"\n"))
			end
			if UI.rootButton then UI.rootButton.Text="RESOLVED VEHICLE ROOT\n"..vehicle.Name.."\n"..safePath(vehicle).."\n[ SHOW ROOT DETAILS ]" end
			if UI.pages.WHEELS.Visible and UI.updateWheels then UI.updateWheels(false) end
			if (UI.pages.VALUES.Visible or UI.pages.ATTRIBUTES.Visible) and UI.updateDataLists then UI.updateDataLists() end
		else
			if UI.liveText then UI.liveText.Text = "Waiting for vehicle...\n\nSit in a Seat or VehicleSeat to begin automatic inspection." end
			if UI.rootButton then UI.rootButton.Text="RESOLVED VEHICLE ROOT\nNONE" end
		end
	end
end)

-- Reveal the finished SPA Tools panel after startup hooks are ready.
finishSPAToolsIntro(spaIntroGui, spaIntroStartedAt)
